Contagem Java

Versão do algoritmo de contagem implementada em Java.

Aguardando dados...
Código Java correspondente: 
import java.util.*;

public class Contagem {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.print("Limite: ");
        double limite = scanner.nextDouble();

        scanner.nextLine();

        System.out.print("Valores separados por vírgula: ");
        String entrada = scanner.nextLine();

        String[] valores = entrada.split(",");

        double menor = Double.MAX_VALUE;

        for (String valor : valores) {
            double numero = Double.parseDouble(valor.trim());

            if (numero < menor) {
                menor = numero;
            }
        }

        int contador = 0;

        System.out.println("Números válidos:");

        for (String valor : valores) {

            double numero =
                Double.parseDouble(valor.trim());

            if (numero >= menor && numero <= limite) {
                System.out.println(numero);
                contador++;
            }
        }

        System.out.println("Total: " + contador);
    }
}

import java.util.*;

public class Fatorial {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.print("Digite N: ");

        int n = scanner.nextInt();

        long fatorial = 1;

        for (int i = 2; i <= n; i++) {
            fatorial *= i;
        }

        System.out.println(
            n + "! = " + fatorial
        );
    }
}

        int n = scanner.nextInt();

        long a = 0;
        long b = 1;

import java.util.*;

public class MDC {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.print("A: ");
        int a = scanner.nextInt();

        System.out.print("B: ");
        int b = scanner.nextInt();

        a = Math.abs(a);
        b = Math.abs(b);

        while (b != 0) {

            int temp = b;

            b = a % b;

            a = temp;
        }

        System.out.println("MDC = " + a);
    }
}

Número Primo Java

Verifica se um número inteiro é primo usando Java.

Aguardando dados...
Código Java correspondente: 
import java.util.*;

public class Primo {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        System.out.print("Digite N: ");

        int n = scanner.nextInt();

        boolean primo = true;

        if (n <= 1) {
            primo = false;
        }

        for (int i = 2; i <= Math.sqrt(n); i++) {

            if (n % i == 0) {
                primo = false;
                break;
            }
        }

        if (primo) {
            System.out.println(
                n + " é primo."
            );
        } else {
            System.out.println(
                n + " não é primo."
            );
        }
    }
}

Ordena uma lista numérica usando o algoritmo Quicksort.

Código Java correspondente: 
import java.util.*;

public class Quicksort {

    public static void quicksort(
        int[] array,
        int inicio,
        int fim
    ) {

        if (inicio < fim) {

            int pivo = array[fim];

            int i = inicio - 1;

            for (int j = inicio; j < fim; j++) {

                if (array[j] <= pivo) {

                    i++;

                    int temp = array[i];

                    array[i] = array[j];

                    array[j] = temp;
                }
            }

            int temp = array[i + 1];

            array[i + 1] = array[fim];

            array[fim] = temp;

            int indicePivo = i + 1;

            quicksort(
                array,
                inicio,
                indicePivo - 1
            );

            quicksort(
                array,
                indicePivo + 1,
                fim
            );
        }
    }

    public static void main(String[] args) {

        int[] numeros = {
            34, 7, 23, 32, 5, 62
        };

        quicksort(
            numeros,
            0,
            numeros.length - 1
        );

        System.out.println(
            Arrays.toString(numeros)
        );
    }
}
